# Dataset cho PyTorch
class PhishingDataset(Dataset):
    def __init__(self, encodings, numerical_features, tfidf_features, labels):
        self.encodings = encodings
        self.numerical_features = numerical_features
        self.tfidf_features = tfidf_features
        self.labels = labels

    def __getitem__(self, idx):
        item = {key: val[idx] for key, val in self.encodings.items()}
        item['numerical_features'] = torch.tensor(self.numerical_features[idx], dtype=torch.float32)
        item['tfidf_features'] = torch.tensor(self.tfidf_features[idx], dtype=torch.float32)
        item['labels'] = torch.tensor(self.labels[idx], dtype=torch.float32)
        return item

    def __len__(self):
        return len(self.labels)

# Hàm vẽ biểu đồ đánh giá
def eval_graph(history):
    acc = history['accuracy']
    val_acc = history['val_accuracy']
    epochs = range(len(acc))
    plt.figure(figsize=(14, 7))
    plt.plot(epochs, acc, 'r', label="Training Accuracy")
    plt.plot(epochs, val_acc, 'b', label="Validation Accuracy")
    plt.legend(loc='upper left')
    plt.title("ACCURACY GRAPH")
    plt.show()

    loss = history['loss']
    val_loss = history['val_loss']
    plt.figure(figsize=(14, 7))
    plt.plot(epochs, loss, 'r', label="Training Loss")
    plt.plot(epochs, val_loss, 'b', label="Validation Loss")
    plt.legend(loc='upper left')
    plt.title("LOSS GRAPH")
    plt.show()

# Hàm vẽ confusion matrix
def conf_matrix(X_test, Y_test, model, is_transformer=False):
    model.eval()
    predictions = []
    with torchmorning
        for batch in X_test:
            if is_transformer:
                input_ids = batch['input_ids'].to(device)
                attention_mask = batch['attention_mask'].to(device)
                numerical_features = batch['numerical_features'].to(device)
                outputs = model(input_ids, attention_mask, numerical_features).squeeze()
            else:
                inputs = batch['tfidf_features'].to(device) if 'tfidf_features' in batch else batch['numerical_features'].to(device)
                outputs = model(inputs).squeeze()
            predictions.extend((outputs >= 0.5).float().cpu().numpy())
    cm = confusion_matrix(Y_test, predictions)
    disp = ConfusionMatrixDisplay(confusion_matrix=cm)
    disp.plot()
    plt.show()
    print(classification_report(Y_test, predictions))

# Định nghĩa mô hình CNN
class CNN(nn.Module):
    def __init__(self, input_size):
        super(CNN, self).__init__()
        self.model = nn.Sequential(
            nn.Conv1d(input_size[1], 16, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(16),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(16, 32, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(32),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(32, 64, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(64),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(64, 128, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(128),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(128, 256, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(256),
            nn.MaxPool1d(2, padding=1),
            nn.Flatten(),
            nn.Dropout(0.3),
            nn.Linear(256 * (input_size[0] // 32), 512),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        return self.model(x)

# Định nghĩa mô hình CNN-LSTM
class CNN_LSTM(nn.Module):
    def __init__(self, input_size):
        super(CNN_LSTM, self).__init__()
        self.cnn = nn.Sequential(
            nn.Conv1d(input_size[1], 16, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(16),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(16, 32, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(32),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(32, 64, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(64),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(64, 128, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(128),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(128, 256, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(256)
        )
        self.lstm = nn.LSTM(256, 128, batch_first=True, bidirectional=False)
        self.fc = nn.Sequential(
            nn.Dropout(0.3),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(128, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        x = self.cnn(x)
        x = x.permute(0, 2, 1)  # Reshape for LSTM
        _, (h_n, _) = self.lstm(x)
        x = h_n.squeeze(0)
        return self.fc(x)

# Định nghĩa mô hình CNN-Transformer
class CNN_Transformer(nn.Module):
    def __init__(self, transformer, num_numerical_features):
        super(CNN_Transformer, self).__init__()
        self.transformer = transformer
        self.cnn = nn.Sequential(
            nn.Conv1d(1, 16, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(16),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(16, 32, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(32),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(32, 64, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(64),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(64, 128, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(128),
            nn.MaxPool1d(2, padding=1),
            nn.Conv1d(128, 256, kernel_size=3, padding='same'),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.BatchNorm1d(256),
            nn.Flatten(),
            nn.Dropout(0.3),
            nn.Linear(256 * (num_numerical_features // 32), 512),
            nn.ReLU()
        )
        self.fc = nn.Sequential(
            nn.Linear(768 + 512, 512),  # 768 from BERT, 512 from CNN
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )

    def forward(self, input_ids, attention_mask, numerical_features):
        transformer_output = self.transformer(input_ids=input_ids, attention_mask=attention_mask).pooler_output
        cnn_output = self.cnn(numerical_features.unsqueeze(-1))
        combined = torch.cat((transformer_output, cnn_output), dim=1)
        return self.fc(combined)

# Định nghĩa mô hình Transformer-LSTM
class Transformer_LSTM(nn.Module):
    def __init__(self, transformer, num_numerical_features):
        super(Transformer_LSTM, self).__init__()
        self.transformer = transformer
        self.lstm = nn.LSTM(num_numerical_features, 128, batch_first=True, bidirectional=False)
        self.fc = nn.Sequential(
            nn.Linear(768 + 128, 512),  # 768 from BERT, 128 from LSTM
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )

    def forward(self, input_ids, attention_mask, numerical_features):
        transformer_output = self.transformer(input_ids=input_ids, attention_mask=attention_mask).pooler_output
        lstm_output, _ = self.lstm(numerical_features.unsqueeze(1))
        lstm_output = lstm_output[:, -1, :]
        combined = torch.cat((transformer_output, lstm_output), dim=1)
        return self.fc(combined)

# Định nghĩa mô hình Transformer
class Transformer(nn.Module):
    def __init__(self, transformer):
        super(Transformer, self).__init__()
        self.transformer = transformer
        self.fc = nn.Sequential(
            nn.Linear(768, 512),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )

    def forward(self, input_ids, attention_mask, numerical_features=None):
        transformer_output = self.transformer(input_ids=input_ids, attention_mask=attention_mask).pooler_output
        return self.fc(transformer_output)

# Hàm huấn luyện mô hình
def train_model(model, train_loader, test_loader, model_name, is_transformer=False, epochs=10):
    model.to(device)
    criterion = nn.BCELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=2e-5 if is_transformer else 1e-3)
    history = {'loss': [], 'val_loss': [], 'accuracy': [], 'val_accuracy': []}

    for epoch in range(epochs):
        model.train()
        total_loss = 0
        correct = 0
        total = 0
        for batch in train_loader:
            if is_transformer:
                input_ids = batch['input_ids'].to(device)
                attention_mask = batch['attention_mask'].to(device)
                numerical_features = batch['numerical_features'].to(device)
                outputs = model(input_ids, attention_mask, numerical_features).squeeze()
            else:
                inputs = batch['tfidf_features'].to(device) if 'tfidf_features' in batch else batch['numerical_features'].to(device)
                outputs = model(inputs).squeeze()
            labels = batch['labels'].to(device)

            optimizer.zero_grad()
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

            total_loss += loss.item()
            predictions = (outputs >= 0.5).float()
            correct += (predictions == labels).sum().item()
            total += labels.size(0)

        avg_loss = total_loss / len(train_loader)
        train_accuracy = correct / total
        history['loss'].append(avg_loss)
        history['accuracy'].append(train_accuracy)

        model.eval()
        val_loss = 0
        correct = 0
        total = 0
        with torch.no_grad():
            for batch in test_loader:
                if is_transformer:
                    input_ids = batch['input_ids'].to(device)
                    attention_mask = batch['attention_mask'].to(device)
                    numerical_features = batch['numerical_features'].to(device)
                    outputs = model(input_ids, attention_mask, numerical_features).squeeze()
                else:
                    inputs = batch['tfidf_features'].to(device) if 'tfidf_features' in batch else batch['numerical_features'].to(device)
                    outputs = model(inputs).squeeze()
                labels = batch['labels'].to(device)

                loss = criterion(outputs, labels)
                val_loss += loss.item()
                predictions = (outputs >= 0.5).float()
                correct += (predictions == labels).sum().item()
                total += labels.size(0)

        avg_val_loss = val_loss / len(test_loader)
        val_accuracy = correct / total
        history['val_loss'].append(avg_val_loss)
        history['val_accuracy'].append(val_accuracy)

        print(f'{model_name} - Epoch {epoch+1}, Loss: {avg_loss:.4f}, Accuracy: {train_accuracy:.4f}, Val Loss: {avg_val_loss:.4f}, Val Accuracy: {val_accuracy:.4f}')

        # Lưu mô hình
        torch.save(model.state_dict(), f'{model_name}.pth')

    return history
